cmake_minimum_required(VERSION 3.24)
project(HangulJpIME LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 17)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/src/service.cpp" service_source)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/src/core.cpp" core_source)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/src/core.h" core_header)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/src/resources/HanBonGo.ico" branding_hash)
string(SHA256 source_hash "${service_source}${core_source}${core_header}${branding_hash}")
set(identity_bytes "")
foreach(i RANGE 0 7)
  math(EXPR offset "${i} * 2")
  string(SUBSTRING "${source_hash}" ${offset} 2 byte)
  string(APPEND identity_bytes "0x${byte},")
endforeach()
configure_file(src/identity.h.in generated/identity.h @ONLY)
configure_file(src/HanBonGo.rc.in generated/HanBonGo.rc @ONLY)
include_directories("${CMAKE_CURRENT_BINARY_DIR}/generated")
add_compile_options(/utf-8 /W4 /EHsc)
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_library(HangulJpIME SHARED src/service.cpp src/core.cpp src/HangulJpIME.def "${CMAKE_CURRENT_BINARY_DIR}/generated/HanBonGo.rc")
target_link_libraries(HangulJpIME PRIVATE ole32 oleaut32 uuid user32 advapi32 gdi32)
add_executable(ime-tool src/tool.cpp src/core.cpp src/integration.cpp)
target_link_libraries(ime-tool PRIVATE ole32 oleaut32 uuid user32 advapi32 shell32)
enable_testing()
add_test(NAME core COMMAND ime-tool --test)
add_test(NAME microsoft-ime COMMAND ime-tool --probe)
# Full TSF activation requires an installed system profile. Run --integration
# after scripts/install.ps1; build-time tests do not register a global input method.
